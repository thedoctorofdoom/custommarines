Class SGMKAllyActor : actor{
	int CheckPlayerinSurroundings(double rad = 256) {
		int ret = 0;
		foreach(pl : players) {
			bool valid = (pl.mo && pl.mo.health > 0 && pl.mo.Distance3D(self) <= rad && self.CheckSight(pl.mo));
			if (valid) { ret++; }
		}
		return ret;
	}
	PlayerPawn GetClosestPlayer() {
		int lastrange = int.max; // or is it int.max() ???
		PlayerPawn closest = null;
		int range = lastrange;
		foreach(pl : players) {
			bool valid = (pl.mo && pl.mo.health > 0);// && self.CheckSight(pl.mo)); // you can remove this if LoS isn't a concern
				if (valid) {
					range = pl.mo.Distance3D(self);// <= rad;
					if (range < lastrange) { closest = pl.mo; }
				}
			}
		return closest;
	}
	void A_FindPlayerTarget(){
		if (target is "PlayerPawn") return;
		target = players[0].mo;//GetClosestPlayer();
		return;
	}
}

Class MarineCommandGuard : Inventory{
	Default{
		Inventory.Amount 1;
		Inventory.MaxAmount 1;
		+INVENTORY.HUBPOWER
		+INVENTORY.PERSISTENTPOWER
		+INVENTORY.UNCLEARABLE
		Tag "Marine Manager";
	}
	override bool Use(bool pickup){
		A_Print("Commanding Marines to Guard.");
		let Marine_Manager = MarineManager(FindInventory("MarineManager"));
		if (Marine_Manager){
			for(int i = 0; i < Marine_Manager.MarineList.Size(); i++){
				if(Marine_Manager.MarineList[i].speed > 0 && Marine_Manager.MarineList[i].bUSESPECIAL){
					State Active = Marine_Manager.MarineList[i].FindState('Active');
					if (Active) Marine_Manager.MarineList[i].SetState(FindState("Active"));
				}
			}
		}
		return false;
	}
	
	States{
	Spawn:
		STLH A -1;
		Stop;
	}
}
